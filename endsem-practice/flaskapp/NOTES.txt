
# POST (Create)

Invoke-RestMethod -Uri 'http://127.0.0.1:5000/users' -Method Post -ContentType 'application/json' -Body '{
	"username": "new_user",
	"email": "new_user@gmail.com"
}'

# PUT (Update)

Invoke-RestMethod -Uri 'http://127.0.0.1:5000/users/1' -Method Put -ContentType 'application/json' -Body '{
	"username": "updated",
	"email": "updated@gmail.com"
}'

# DELETE (Delete)

Invoke-RestMethod -Uri 'http://127.0.0.1:5000/users/1' -Method Delete -ContentType 'application/json' -Body '{
	"username": "updated",
	"email": "updated@gmail.com"
}'
